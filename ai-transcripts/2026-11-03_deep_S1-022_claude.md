import type { NormalizedReview, Review } from "@/types/review";
import { InvalidReviewError, TargetNotFoundError } from "./errors";


/** Data-access port. The Supabase implementation must use the user-scoped client (RLS on). */
export interface ReviewStore {
  getTargetOwnerId(targetId: string): Promise<string | null>;
  insertReviews(targetId: string, reviews: NormalizedReview[]): Promise<Review[]>;
  listReviews(targetId: string): Promise<Review[]>;
}


async function assertOwnsTarget(store: ReviewStore, userId: string, targetId: string) {
  const ownerId = await store.getTargetOwnerId(targetId);


  if (ownerId === null || ownerId !== userId) {
    throw new TargetNotFoundError();
  }
}




function assertValid(review: NormalizedReview) {
  if (typeof review.reviewText !== "string" || review.reviewText.trim() === "") {
    throw new InvalidReviewError("Review text is required.");
  }


  if (typeof review.rating !== "number" || !Number.isFinite(review.rating)) {
    throw new InvalidReviewError("Review rating must be a number.");
  }
}


/**
 * userId MUST come from the verified session (S1-011), never from the request body (S1-BR-010).
 * Persists nothing if any review is invalid, so a bad batch is never partially saved.
 */
export async function saveReviews(
  store: ReviewStore,
  userId: string,
  targetId: string,
  reviews: NormalizedReview[],
): Promise<Review[]> {
  await assertOwnsTarget(store, userId, targetId);
  reviews.forEach(assertValid);


  if (reviews.length === 0) {
    return [];
  }


  return store.insertReviews(targetId, reviews);
}


export async function getReviewsForTarget(
  store: ReviewStore,


 userId: string,
  targetId: string,
): Promise<Review[]> {
  await assertOwnsTarget(store, userId, targetId);


  return store.listReviews(targetId);
}

import type { SupabaseClient } from "@supabase/supabase-js";
import type { NormalizedReview, Review } from "@/types/review";
import type { ReviewStore } from "./review-service";


interface ReviewRow {
  id: string;
  analysis_target_id: string;
  review_text: string;
  rating: number;
  source_review_id: string | null;
  reviewer_name: string | null;
  reviewed_at: string | null;
  review_url: string | null;
  created_at: string;
}


function toReview(row: ReviewRow): Review {
  return {
    id: row.id,
    analysisTargetId: row.analysis_target_id,
    reviewText: row.review_text,
    rating: Number(row.rating),


   sourceReviewId: row.source_review_id,
    reviewerName: row.reviewer_name,
    reviewedAt: row.reviewed_at,
    reviewUrl: row.review_url,
    createdAt: row.created_at,
  };
}


/** Pass the user-scoped server client (cookie session). Never the service-role client. */
export function createSupabaseReviewStore(client: SupabaseClient): ReviewStore {
  return {
    async getTargetOwnerId(targetId) {
      const { data, error } = await client
        .from("analysis_targets")
        .select("owner_id")
        .eq("id", targetId)
        .maybeSingle();


      if (error) {
        throw new Error("Failed to load analysis target.");
      }


      return data?.owner_id ?? null;
    },


    async insertReviews(targetId, reviews: NormalizedReview[]) {
      const { data, error } = await client
        .from("reviews")
        .insert(
          reviews.map((review) => ({
            analysis_target_id: targetId,
            review_text: review.reviewText,


           rating: review.rating,
            source_review_id: review.sourceReviewId ?? null,
            reviewer_name: review.reviewerName ?? null,
            reviewed_at: review.reviewedAt ?? null,
            review_url: review.reviewUrl ?? null,
          })),
        )
        .select();


      if (error) {
        throw new Error("Failed to save reviews.");
      }


      return (data as ReviewRow[]).map(toReview);
    },


    async listReviews(targetId) {
      const { data, error } = await client
        .from("reviews")
        .select()
        .eq("analysis_target_id", targetId)
        .order("created_at", { ascending: true });


      if (error) {
        throw new Error("Failed to load reviews.");
      }


      return (data as ReviewRow[]).map(toReview);
    },
  };
}

import { getReviewsForTarget, saveReviews, type ReviewStore } from "@/services/reviews/review-service";
import { InvalidReviewError, TargetNotFoundError } from "@/services/reviews/errors";
import type { NormalizedReview, Review } from "@/types/review";


function createInMemoryStore(targetOwners: Record<string, string>): ReviewStore {
  const rows: Review[] = [];
  let nextId = 1;


  return {
    async getTargetOwnerId(targetId) {
      return targetOwners[targetId] ?? null;
    },
    async insertReviews(targetId, reviews) {
      const created = reviews.map((review) => ({
        ...review,
        id: `review-${nextId++}`,
        analysisTargetId: targetId,
        createdAt: new Date().toISOString(),
      }));
      rows.push(...created);


      return created;
    },
    async listReviews(targetId) {
      return rows.filter((row) => row.analysisTargetId === targetId);
    },
  };
}


const userA = "user-a";


const userB = "user-b";
const reviewsA: NormalizedReview[] = [
  { reviewText: "Great food", rating: 5, sourceReviewId: "src-1", reviewerName: "Sam" },
  { reviewText: "Slow service", rating: 2 },
];
const reviewsB: NormalizedReview[] = [{ reviewText: "Fine", rating: 3 }];


describe("review persistence (S1-022)", () => {
  let store: ReviewStore;


  beforeEach(() => {
    store = createInMemoryStore({ "target-a": userA, "target-b": userB, "target-a2": userA });
  });


  test("persists reviews after successful ingestion and they can be retrieved", async () => {
    const saved = await saveReviews(store, userA, "target-a", reviewsA);
    const retrieved = await getReviewsForTarget(store, userA, "target-a");


    expect(saved).toHaveLength(2);
    expect(retrieved).toHaveLength(2);
    expect(retrieved.every((review) => review.id)).toBe(true);
  });

 });
