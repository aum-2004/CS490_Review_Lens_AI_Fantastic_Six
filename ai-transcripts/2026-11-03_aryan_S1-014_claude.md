const SOURCE_URL_UNSUPPORTED_MESSAGE =

const GOOGLE_MAPS_HOSTNAMES = new Set(["google.com", "www.google.com", "maps.google.com"]);
const GOOGLE_MAPS_PLACE_IDENTIFIER_PARAMETERS = ["cid", "place_id", "query_place_id"];

export const SUPPORTED_REVIEW_PLATFORM = "google_maps" as const;

export type SupportedReviewPlatform = typeof SUPPORTED_REVIEW_PLATFORM;

function hasPlaceIdentifier(url: URL): boolean {
  return GOOGLE_MAPS_PLACE_IDENTIFIER_PARAMETERS.some((parameter) => {
    const value = url.searchParams.get(parameter);
    return value !== null && value.trim().length > 0;
  });
}

function hasMeaningfulPlacePath(path: string): boolean {
  const placePathPrefix = "/maps/place/";

  if (!path.startsWith(placePathPrefix)) {
    return false;
  }

  const firstPlaceSegment = path.slice(placePathPrefix.length).split("/", 1)[0];

  try {
    const decodedPlaceSegment = decodeURIComponent(firstPlaceSegment);
    return decodedPlaceSegment.replaceAll("/", "").trim().length > 0;
  } catch {
    return false;
  }
}

function isGoogleMapsBusinessUrl(url: URL): boolean {
  if (!GOOGLE_MAPS_HOSTNAMES.has(url.hostname) || url.username || url.password || url.port) {
    return false;
  }

  const path = url.pathname.toLowerCase();

  if (hasMeaningfulPlacePath(path)) {
    return true;
  }

  const acceptsBusinessQuery =
    path === "/maps" ||
    path === "/maps/" ||
    path === "/maps/search" ||
    path === "/maps/search/" ||
    (url.hostname === "maps.google.com" && path === "/");

  return acceptsBusinessQuery && hasPlaceIdentifier(url);
}

/**
 * Validates and normalizes data for one Google Maps analysis target.
 *
 * S1-015 integration contract: a protected server boundary must call this function with the
 * parsed request body and return the failure as HTTP 400 before invoking persistence. On success,
 * it may pass only `result.data` to the creation service and must derive ownership independently
 * from the authenticated session.
 *
 * Shortened URLs such as maps.app.goo.gl are intentionally unsupported until S1-020 confirms that
 * the ingestion provider can use them without redirect resolution.
 */
export function validateAnalysisTargetInput(input: unknown): AnalysisTargetValidationResult {
  if (!isRecord(input)) {
    return failure([
      {
        code: "INVALID_INPUT_TYPE",
        message: "Target information must be an object.",
      },
    ]);
  }

  const issues: AnalysisTargetValidationIssue[] = [];
  const rawName = hasOwnProperty(input, "name") ? input.name : undefined;
  const rawSourceUrl = hasOwnProperty(input, "sourceUrl") ? input.sourceUrl : undefined;

  let normalizedName: string | undefined;
  let normalizedSourceUrl: string | undefined;

  if (rawName === undefined || rawName === null) {
    issues.push({
      field: "name",
      code: "TARGET_NAME_REQUIRED",
      message: TARGET_NAME_REQUIRED_MESSAGE,
    });
  } else if (typeof rawName !== "string") {
    issues.push({
      field: "name",
      code: "TARGET_NAME_INVALID_TYPE",
      message: TARGET_NAME_REQUIRED_MESSAGE,
    });
  } else {
    normalizedName = rawName.trim();

    if (normalizedName.length === 0) {
      issues.push({
        field: "name",
        code: "TARGET_NAME_REQUIRED",
        message: TARGET_NAME_REQUIRED_MESSAGE,
      });
    } else if (characterCount(normalizedName) > MAX_TARGET_NAME_LENGTH) {
      issues.push({
        field: "name",
        code: "TARGET_NAME_TOO_LONG",
        message: TARGET_NAME_TOO_LONG_MESSAGE,
      });
    }
  }

  if (rawSourceUrl === undefined || rawSourceUrl === null) {
    issues.push({
      field: "sourceUrl",
      code: "SOURCE_URL_REQUIRED",
      message: SOURCE_URL_REQUIRED_MESSAGE,
    });
  } else if (typeof rawSourceUrl !== "string") {
    issues.push({
      field: "sourceUrl",
      code: "SOURCE_URL_INVALID_TYPE",
      message: SOURCE_URL_INVALID_MESSAGE,
    });
  } else {
    normalizedSourceUrl = rawSourceUrl.trim();

    if (normalizedSourceUrl.length === 0) {
      issues.push({
        field: "sourceUrl",
        code: "SOURCE_URL_REQUIRED",
        message: SOURCE_URL_REQUIRED_MESSAGE,
      });
    } else {
      let parsedUrl: URL | undefined;

      try {
        parsedUrl = new URL(normalizedSourceUrl);
      } catch {
        issues.push({
          field: "sourceUrl",
          code: "SOURCE_URL_MALFORMED",
          message: SOURCE_URL_INVALID_MESSAGE,
        });
      }

      if (parsedUrl) {
        if (parsedUrl.protocol !== "http:" && parsedUrl.protocol !== "https:") {
          issues.push({
            field: "sourceUrl",
            code: "SOURCE_URL_UNSUPPORTED_PROTOCOL",
            message: SOURCE_URL_INVALID_MESSAGE,
          });
        } else if (!isGoogleMapsBusinessUrl(parsedUrl)) {
          issues.push({
            field: "sourceUrl",
            code: "SOURCE_URL_UNSUPPORTED_PLATFORM",
            message: SOURCE_URL_UNSUPPORTED_MESSAGE,
          });
        }
      }
    }
  }

  if (issues.length > 0 || normalizedName === undefined || normalizedSourceUrl === undefined) {
    return failure(issues);
  }

  return {
    success: true,
    data: {
      name: normalizedName,
      sourceUrl: normalizedSourceUrl,
      platform: SUPPORTED_REVIEW_PLATFORM,
    },
  };
}

export function getSupabasePublicConfig() {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL?.trim();
  const publishableKey = process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY?.trim();

  if (!url || !publishableKey) {
    const missing: string[] = [];

    if (!url) {
      missing.push("NEXT_PUBLIC_SUPABASE_URL");
    }

    if (!publishableKey) {
      missing.push("NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY");
    }

    throw new Error(
      `Missing required Supabase environment variables: ${missing.join(", ")}. Check .env.local.`,
    );
  }

  let parsedUrl: URL;

  try {
    parsedUrl = new URL(url);
  } catch {
    throw new Error("Invalid NEXT_PUBLIC_SUPABASE_URL. Expected a valid HTTP or HTTPS URL.");
  }

  if (parsedUrl.protocol !== "https:" && parsedUrl.protocol !== "http:") {
    throw new Error("Invalid NEXT_PUBLIC_SUPABASE_URL. Expected a valid HTTP or HTTPS URL.");
  }

  return {
    url,
    publishableKey,
  };
}

-- S1-014: AnalysisTarget persistence model.
--
-- One row per business or item a user wants to analyze. Every target belongs to exactly one
-- authenticated user (S1-003, engineering standards section 8). Reviews and ingestion attempts
-- will reference analysis_targets.id in later stories, so a target can have many of each.

create table public.analysis_targets (
  id uuid primary key default gen_random_uuid(),
  owner_id uuid not null references auth.users (id) on delete cascade,
  name text not null,
  platform text not null,
  source_url text not null,
  created_at timestamptz not null default now(),

  -- Mirrors the S1-016 validation rules so the database rejects bad rows even if a caller
  -- skips validation.
  constraint analysis_targets_name_present check (char_length(btrim(name)) > 0),
  constraint analysis_targets_name_length check (char_length(name) <= 120),
  constraint analysis_targets_platform_supported check (platform in ('google_maps')),
  constraint analysis_targets_source_url_http check (source_url ~* '^https?://')
);

comment on table public.analysis_targets is

comment on column public.analysis_targets.platform is

import type { SupabaseClient } from "@supabase/supabase-js";

// S1-016 (#32) defines the same platform value in src/lib/validation/analysis-target.ts.
// Import it from there once #32 is merged so there is a single definition.
export type ReviewPlatform = "google_maps";

export interface AnalysisTarget {
  id: string;
  ownerId: string;
  name: string;
  platform: ReviewPlatform;
  sourceUrl: string;
  createdAt: string;
}

/** Already-validated target fields. Ownership is never part of this input. */
export interface NewAnalysisTarget {
  name: string;
  platform: ReviewPlatform;
  sourceUrl: string;
}

interface AnalysisTargetRow {
  id: string;
  owner_id: string;
  name: string;
  platform: ReviewPlatform;
  source_url: string;
  created_at: string;
}

const TABLE = "analysis_targets";
const COLUMNS = "id, owner_id, name, platform, source_url, created_at";
const UUID_PATTERN = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

export class AnalysisTargetPersistenceError extends Error {
  readonly code = "ANALYSIS_TARGET_PERSISTENCE_FAILED";

  constructor(operation: "create" | "get", options?: { cause?: unknown }) {
    // The message is safe to show; database details stay on `cause` for server-side logging.
    super(`Could not ${operation} the analysis target.`, options);
    this.name = "AnalysisTargetPersistenceError";
  }
}

function toAnalysisTarget(row: AnalysisTargetRow): AnalysisTarget {
  return {
    id: row.id,
    ownerId: row.owner_id,
    name: row.name,
    platform: row.platform,
    sourceUrl: row.source_url,
    createdAt: row.created_at,
  };
}

/**
 * Persists a new AnalysisTarget owned by `ownerId`.
 *
 * `ownerId` must come from the verified session (S1-011), never from request input. The client
 * must be a user-scoped Supabase client so Row Level Security also checks ownership.
 */
export async function createAnalysisTarget(
  client: SupabaseClient,
  ownerId: string,
  input: NewAnalysisTarget,
): Promise<AnalysisTarget> {
  const { data, error } = await client
    .from(TABLE)
    .insert({
      owner_id: ownerId,
      name: input.name,
      platform: input.platform,
      source_url: input.sourceUrl,
    })
    .select(COLUMNS)
    .single<AnalysisTargetRow>();

  if (error || !data) {
    throw new AnalysisTargetPersistenceError("create", { cause: error });
  }

  return toAnalysisTarget(data);
}

/**
 * Returns the target with `targetId` if it belongs to `ownerId`, otherwise `null`.
 *
 * Missing and foreign targets both return `null` so callers can respond with 404 without
 * revealing that another user's target exists.
 */
export async function getAnalysisTargetById(
  client: SupabaseClient,
  ownerId: string,
  targetId: string,
): Promise<AnalysisTarget | null> {
  if (!UUID_PATTERN.test(targetId)) {
    return null;
  }

  const { data, error } = await client
    .from(TABLE)
    .select(COLUMNS)
    .eq("id", targetId)
    .eq("owner_id", ownerId)
    .maybeSingle<AnalysisTargetRow>();

  if (error) {
    throw new AnalysisTargetPersistenceError("get", { cause: error });
  }

  return data ? toAnalysisTarget(data) : null;
}

-- Test-only stand-in for the parts of Supabase that migrations depend on, so they can run
-- against a plain local PostgreSQL. Never apply this to a real Supabase project: Supabase
-- already provides the auth schema, auth.uid(), and these roles.

do $$
begin
  if not exists (select 1 from pg_roles where rolname = 'anon') then
    create role anon nologin;
  end if;
  if not exists (select 1 from pg_roles where rolname = 'authenticated') then
    create role authenticated nologin;
  end if;
end
$$;

grant anon, authenticated to current_user;

create schema auth;
create table auth.users (id uuid primary key);

-- Same contract as Supabase: the signed-in user's id from the request JWT, or null.
create function auth.uid() returns uuid
language sql stable
as $$
  select nullif(current_setting('request.jwt.claims', true), '')::jsonb ->> 'sub'
$$;

grant usage on schema auth to anon, authenticated;
grant execute on function auth.uid() to anon, authenticated;

-- Supabase grants table access to these roles by default and relies on RLS to restrict rows.
alter default privileges in schema public grant all on tables to anon, authenticated;

### Responsibilities

**Frontend**

- `src/app/` — URL routes. A folder with a `page.tsx` becomes a page; `layout.tsx` wraps pages.
- `src/components/` — Reusable React components. shadcn/ui primitives go in `ui/`;
  feature components go in folders like `analysis-targets/`; shared ones in `shared/`.

**Backend**

- `src/app/api/` — Route Handlers. Each `route.ts` is an HTTP endpoint. Handlers stay thin:
  validate input, authenticate, authorize, call a service, return the shared response shape.
- `src/services/` — Business logic and data access, organized by domain (`auth`,
  `analysis-targets`, `ingestion`, `reviews`). Server-only.

**Data / persistent storage**

- `src/lib/supabase/` — Supabase client configuration. Reserved; nothing connects to a database
  yet.
- `.env.example` / `.env.local` — Connection settings. Values are never hard-coded in source.
- `supabase/migrations/` — Database schema (to be added by S1-014).

**Shared code**

- `src/lib/` — Framework clients and small utilities (`supabase/`, and later `validation/`,
  `utils/`).
- `src/types/` — Shared TypeScript types, created when first needed.

Import from `src` using the `@/` alias, for example `import { x } from "@/lib/utils/x"`.

Folders are created by the stories that need them; each reserved folder contains a short
`README.md` describing what belongs there. The full conventions (naming, error handling,
response shapes, testing) are in `docs/engineering-standards.md` (S1-001).
