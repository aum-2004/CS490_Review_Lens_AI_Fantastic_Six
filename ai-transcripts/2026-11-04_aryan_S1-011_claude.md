````text
import type { SupabaseClient } from "@supabase/supabase-js";

// S1-016 (#32) defines the same platform value in src/lib/validation/analysis-target.ts.
// Import it from there once #32 is merged so there is a single definition.
export type ReviewPlatform = "google_maps";

export interface AnalysisTarget {
  id: string;
  ownerId: string;
  name: string;
  platform: ReviewPlatform;
  sourceUrl: string;
  /** Google place ID resolved during ingestion (S1-020); null until then. */
  googlePlaceId: string | null;
  createdAt: string;
}

/** Already-validated target fields. Ownership is never part of this input. */
export interface NewAnalysisTarget {
  name: string;
  platform: ReviewPlatform;
  sourceUrl: string;
  googlePlaceId?: string;
}

interface AnalysisTargetRow {
  id: string;
  owner_id: string;
  name: string;
  platform: ReviewPlatform;
  source_url: string;
  google_place_id: string | null;
  created_at: string;
}

const TABLE = "analysis_targets";
const COLUMNS = "id, owner_id, name, platform, source_url, google_place_id, created_at";
const UUID_PATTERN = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

export class AnalysisTargetPersistenceError extends Error {
  readonly code = "ANALYSIS_TARGET_PERSISTENCE_FAILED";

  constructor(operation: "create" | "get", options?: { cause?: unknown }) {
    // The message is safe to show; database details stay on `cause` for server-side logging.
    super(`Could not ${operation} the analysis target.`, options);
    this.name = "AnalysisTargetPersistenceError";
  }
}

function toAnalysisTarget(row: AnalysisTargetRow): AnalysisTarget {
  return {
    id: row.id,
    ownerId: row.owner_id,
    name: row.name,
    platform: row.platform,
    sourceUrl: row.source_url,
    googlePlaceId: row.google_place_id,
    createdAt: row.created_at,
  };
}

/**
 * Persists a new AnalysisTarget owned by `ownerId`.
 *
 * `ownerId` must come from the verified session (S1-011), never from request input. The client
 * must be a user-scoped Supabase client so Row Level Security also checks ownership.
 */
export async function createAnalysisTarget(
  client: SupabaseClient,
  ownerId: string,
  input: NewAnalysisTarget,
): Promise<AnalysisTarget> {
  const { data, error } = await client
    .from(TABLE)
    .insert({
      owner_id: ownerId,
      name: input.name,
      platform: input.platform,
      source_url: input.sourceUrl,
      google_place_id: input.googlePlaceId ?? null,
    })
    .select(COLUMNS)
    .single<AnalysisTargetRow>();

  if (error || !data) {
    throw new AnalysisTargetPersistenceError("create", { cause: error });
  }

  return toAnalysisTarget(data);
}

/**
 * Returns the target with `targetId` if it belongs to `ownerId`, otherwise `null`.
 *
 * Missing and foreign targets both return `null` so callers can respond with 404 without
 * revealing that another user's target exists.
 */
export async function getAnalysisTargetById(
  client: SupabaseClient,
  ownerId: string,
  targetId: string,
): Promise<AnalysisTarget | null> {
  if (!UUID_PATTERN.test(targetId)) {
    return null;
  }

  const { data, error } = await client
    .from(TABLE)
    .select(COLUMNS)
    .eq("id", targetId)
    .eq("owner_id", ownerId)
    .maybeSingle<AnalysisTargetRow>();

  if (error) {
    throw new AnalysisTargetPersistenceError("get", { cause: error });
  }

  return data ? toAnalysisTarget(data) : null;
}

-- S1-014: AnalysisTarget persistence model.
--
-- One row per business or item a user wants to analyze. Every target belongs to exactly one
-- authenticated user (S1-003, engineering standards section 8). Reviews and ingestion attempts
-- will reference analysis_targets.id in later stories, so a target can have many of each.

create table public.analysis_targets (
  id uuid primary key default gen_random_uuid(),
  owner_id uuid not null references auth.users (id) on delete cascade,
  name text not null,
  platform text not null,
  source_url text not null,
  google_place_id text,
  created_at timestamptz not null default now(),
````

````text
Route Groups are a folder convention that let you organize routes by category or team.

## Convention

A route group can be created by wrapping a folder's name in parenthesis: `(folderName)`.

This convention indicates the folder is for organizational purposes and should **not be included** in the route's URL path.

<Image
  alt="An example folder structure using route groups"
  srcLight="/docs/light/project-organization-route-groups.png"
  srcDark="/docs/dark/project-organization-route-groups.png"
  width="1600"
  height="849"
/>

## Use cases

- Organizing routes by team, concern, or feature.
- Defining multiple [root layouts](/docs/app/api-reference/file-conventions/layout#root-layout).
- Opting specific route segments into sharing a layout, while keeping others out.

## Caveats

- **Full page load**: If you navigate between routes that use different root layouts, it'll trigger a full page reload. For example, navigating from `/cart` that uses `app/(shop)/layout.js` to `/blog` that uses `app/(marketing)/layout.js`. This **only** applies to multiple root layouts.
- **Conflicting paths**: Routes in different groups should not resolve to the same URL path. For example, `(marketing)/about/page.js` and `(shop)/about/page.js` would both resolve to `/about` and cause an error.
- **Top-level root layout**: If you use multiple root layouts without a top-level `layout.js` file, make sure your home route (/) is defined within one of the route groups, e.g. app/(marketing)/page.js.

# AnalysisTarget creation integration contract

S1-015 intentionally does not implement authentication (S1-009) or persistence (S1-014).
The client form submits only `name` and `sourceUrl` to `POST /api/analysis-targets`; it never sends
an owner or user identifier.

When the dependent stories are available, the protected Route Handler must:

1. Resolve and verify the Supabase session. Return `401` before any write when no authenticated
   identity exists.
2. Parse the JSON body and call `validateAnalysisTargetInput()` from
   `src/lib/validation/analysis-target.ts`. Return its structured error as `400` before calling
   persistence.
3. Pass only the validated `name`, `sourceUrl`, and fixed `platform`, plus the server-derived user
   identity, to the S1-014 service.
4. Return `201` as `{ success: true, data: AnalysisTarget }` after a genuine committed write.
5. Translate expected failures to the shared safe error response and never return fabricated target
   data.
````

````text
### Convention

Route Handlers are defined in a [`route.js|ts` file](/docs/app/api-reference/file-conventions/route) inside the `app` directory:

```

Route Handlers can be nested anywhere inside the `app` directory, similar to `page.js` and `layout.js`. But there **cannot** be a `route.js` file at the same route segment level as `page.js`.

### Supported HTTP Methods

The following [HTTP methods](https://developer.mozilla.org/docs/Web/HTTP/Methods) are supported: `GET`, `POST`, `PUT`, `PATCH`, `DELETE`, `HEAD`, and `OPTIONS`. If an unsupported method is called, Next.js will return a `405 Method Not Allowed` response.

### Extended `NextRequest` and `NextResponse` APIs

In addition to supporting the native [Request](https://developer.mozilla.org/docs/Web/API/Request) and [Response](https://developer.mozilla.org/docs/Web/API/Response) APIs, Next.js extends them with [`NextRequest`](/docs/app/api-reference/functions/next-request) and [`NextResponse`](/docs/app/api-reference/functions/next-response) to provide convenient helpers for advanced use cases.

### Caching

Route Handlers are not cached by default. You can, however, opt into caching for `GET` methods. Other supported HTTP methods are **not** cached. # S1-011: Ownership from authenticated identity
````

````text
## Decision

ReviewLens stores the Supabase Auth user ID (`auth.users.id`) as the owner of user-owned records.
This is the "provider user identifier stored as owner" approach from the story. There is no separate
internal User table, because nothing in Sprint 1 needs profile data that Supabase Auth doesn't hold.

Ownership is enforced at two independent layers:

1. **Application:** the Route Handler takes the owner only from `requireApiUser()`, which verifies
   the session with Supabase Auth on each request.
2. **Database:** Row Level Security requires `auth.uid() = owner_id`, using a client that carries the
   user's session. A service-role client bypasses this and must not be used for user requests.

## Rules for user-owned records

- Never read an owner from the request. `ownerId`, `userId`, and similar body fields are ignored, not
  rejected, so a spoofed value has no effect.
- Pass only validated fields to persistence. Repository inputs don't include ownership, so it remains
  a separate argument supplied by the server.
- Authenticate before reading the request body or creating a database client.
- `AnalysisTarget` stores `owner_id` directly. Reviews and ingestion attempts inherit ownership through
  their parent target, as the engineering standards (section 8) describe. S1-013 and S1-022 own that
  parent check and its tests.
````
